Plot activation functions and derivatives with Python

Plotting a function beside its derivative connects forward representations with backward gradients.

WJunhao WeiAbout 2 min read

Plotting a function beside its derivative connects forward representations with backward gradients. This script samples five activations over a fixed interval, uses consistent colors across both panels, and saves a scalable vector figure. Expressions, derivatives, and axis ranges are preserved for reproducibility.

Each function uses the same color in both panels. SVG output scales with the page. The script explicitly chooses a plotted derivative value for ReLU at zero; this convention does not make the function mathematically differentiable there.

Function values and derivatives

Reproduction

Run the script in a Python environment with NumPy and Matplotlib. It writes activation_functions.svg in the same directory. ReLU is nondifferentiable at zero; the figure uses zero as a plotting convention at that point.

Complete code

from pathlib import Path

import matplotlib.pyplot as plt
import numpy as np
from math import erf


def normal_cdf(x: np.ndarray) -> np.ndarray:
    return 0.5 * (1.0 + np.vectorize(erf)(x / np.sqrt(2.0)))


x = np.linspace(-6.0, 6.0, 1201)
sigmoid = 1.0 / (1.0 + np.exp(-x))
tanh = np.tanh(x)
relu = np.maximum(x, 0.0)
phi = np.exp(-0.5 * x**2) / np.sqrt(2.0 * np.pi)
gelu = x * normal_cdf(x)
silu = x * sigmoid

curves = {
    "Sigmoid": (sigmoid, sigmoid * (1.0 - sigmoid)),
    "Tanh": (tanh, 1.0 - tanh**2),
    "ReLU": (relu, (x > 0.0).astype(float)),
    "GELU": (gelu, normal_cdf(x) + x * phi),
    "SiLU": (silu, sigmoid + x * sigmoid * (1.0 - sigmoid)),
}

colors = {
    "Sigmoid": "#3366CC",
    "Tanh": "#109618",
    "ReLU": "#DC3912",
    "GELU": "#990099",
    "SiLU": "#FF9900",
}

plt.rcParams.update(
    {
        "font.size": 10,
        "axes.titlesize": 12,
        "axes.labelsize": 10,
        "legend.fontsize": 9,
        "svg.fonttype": "none",
    }
)

fig, axes = plt.subplots(1, 2, figsize=(11.2, 4.2), constrained_layout=True)

for name, (value, derivative) in curves.items():
    axes[0].plot(x, value, label=name, color=colors[name], linewidth=2.0)
    axes[1].plot(x, derivative, label=name, color=colors[name], linewidth=2.0)

for axis in axes:
    axis.axhline(0.0, color="#666666", linewidth=0.8)
    axis.axvline(0.0, color="#666666", linewidth=0.8)
    axis.grid(True, color="#D9D9D9", linewidth=0.7, alpha=0.8)
    axis.set_xlim(-6.0, 6.0)
    axis.set_xlabel("Input x")

axes[0].set_title("Activation values")
axes[0].set_ylabel("f(x)")
axes[0].set_ylim(-2.0, 6.0)
axes[1].set_title("Activation derivatives")
axes[1].set_ylabel("f'(x)")
axes[1].set_ylim(-0.2, 1.2)
axes[1].legend(loc="upper left", ncol=2, frameon=True)

output = Path(__file__).with_name("activation_functions.svg")
fig.savefig(output, format="svg", bbox_inches="tight")
print(output)
